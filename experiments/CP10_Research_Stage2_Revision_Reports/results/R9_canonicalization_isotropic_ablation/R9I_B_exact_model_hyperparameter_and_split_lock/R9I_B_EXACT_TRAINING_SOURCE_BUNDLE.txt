138: # POSITIONAL ENCODING
139: # ============================================================
140: 
141: class PositionalEncoding(nn.Module):
142: 
143:     def __init__(
144:         self,
145:         d_model,
146:         max_len=100
147:     ):
148: 
149:         super().__init__()
150: 
151: 
152:         pe = torch.zeros(
153:             max_len,
154:             d_model
155:         )
156: 
157: 
158:         position = torch.arange(
159:             max_len,
160:             dtype=torch.float32
161:         ).unsqueeze(
162:             1
163:         )
164: 
165: 
166:         div_term = torch.exp(
167: 
168:             torch.arange(
169:                 0,
170:                 d_model,
171:                 2,
172:                 dtype=torch.float32
173:             )
174: 
175:             *
176:             (
177:                 -math.log(10000.0)
178:                 /
179:                 d_model
180:             )
181:         )
182: 
183: 
184:         pe[:, 0::2] = torch.sin(
185:             position
186:             *
187:             div_term
188:         )
189: 
190: 
191:         pe[:, 1::2] = torch.cos(
192:             position
193:             *
194:             div_term
195:         )
196: 
197: 
198:         self.register_buffer(
199:             "pe",
200:             pe.unsqueeze(0)
201:         )
202: 
203: 
204:     def forward(self, x):
205: 
206:         return (
207: 
208:             x
209: 
210:             +
211: 
212:             self.pe[
213:                 :,
214:                 :x.size(1)
215:             ]
216: 
217:         )

224: class TrajectoryTransformer(nn.Module):
225: 
226:     def __init__(
227:         self,
228:         future_length
229:     ):
230: 
231:         super().__init__()
232: 
233: 
234:         self.input_projection = nn.Linear(
235:             2,
236:             D_MODEL
237:         )
238: 
239: 
240:         self.position = PositionalEncoding(
241:             D_MODEL,
242:             max_len=32

245: 
246:         encoder_layer = (
247:             nn.TransformerEncoderLayer(
248: 
249:                 d_model=D_MODEL,
250: 
251:                 nhead=NHEAD,
252: 
253:                 dim_feedforward=DIM_FEEDFORWARD,
254: 
255:                 dropout=DROPOUT,
256: 
257:                 activation="gelu",
258: 
259:                 batch_first=True,
260: 
261:                 norm_first=True
262:             )
263:         )
264: 
265: 
266:         self.encoder = nn.TransformerEncoder(
267: 
268:             encoder_layer,
269: 
270:             num_layers=NUM_LAYERS
271:         )
272: 
273: 
274:         self.norm = nn.LayerNorm(
275:             D_MODEL
276:         )
277: 
278: 
279:         self.head = nn.Sequential(
280: 
281:             nn.Linear(
282:                 D_MODEL,
283:                 D_MODEL
284:             ),
285: 
286:             nn.GELU(),
287: 
288:             nn.Dropout(
289:                 DROPOUT
290:             ),
291: 
292:             nn.Linear(
293:                 D_MODEL,
294:                 future_length * 2
295:             )
296:         )
297: 
298: 
299:         self.future_length = (
300:             future_length
301:         )
302: 
303: 
304:     def forward(self, x):
305: 
306: 
307:         x = self.input_projection(
308:             x
309:         )
310: 
311: 
312:         x = self.position(
313:             x
314:         )
315: 
316: 
317:         x = self.encoder(
318:             x
319:         )
320: 
321: 
322:         # Mean pooling over the observed trajectory
323:         x = x.mean(
324:             dim=1
325:         )
326: 
327: 
328:         x = self.norm(
329:             x
330:         )
331: 
332: 
333:         output = self.head(
334:             x
335:         )
336: 
337: 
338:         return output.view(
339: 
340:             -1,
341: 
342:             self.future_length,
343: 
344:             2
345:         )

349: # LOSS
350: # ============================================================
351: 
352: def weighted_smooth_l1(
353:     prediction,
354:     target,
355:     weights
356: ):
357: 
358: 
359:     element_loss = (
360:         nn.functional.smooth_l1_loss(
361: 
362:             prediction,
363: 
364:             target,
365: 
366:             reduction="none"
367: 
368:         )
369:     )
370: 
371: 
372:     sample_loss = element_loss.mean(
373:         dim=(1, 2)
374:     )
375: 
376: 
377:     return (
378: 
379:         (
380:             sample_loss
381:             *
382:             weights
383:         ).sum()
384: 
385:         /
386:         weights.sum()
387: 
388:     )

392: # INVERSE TRANSFORM
393: # ============================================================
394: 
395: def inverse_transform_batch(
396:     canonical,
397:     origin,
398:     angle,
399:     scale
400: ):
401: 
402: 
403:     output = []
404: 
405: 
406:     for i in range(
407:         len(canonical)
408:     ):
409: 
410: 
411:         a = float(
412:             angle[i]
413:         )
414: 
415: 
416:         c = np.cos(a)
417: 
418:         s = np.sin(a)
419: 
420: 
421:         R = np.array(
422:             [
423:                 [c, s],
424:                 [-s, c]
425:             ],
426:             dtype=np.float64
427:         )
428: 
429: 
430:         local = (
431: 
432:             canonical[i]
433: 
434:             *
435:             float(
436:                 scale[i]
437:             )
438: 
439:         ) @ R
440: 
441: 
442:         absolute = (
443: 
444:             local
445: 
446:             +
447:             origin[i]
448: 
449:         )
450: 
451: 
452:         output.append(
453:             absolute
454:         )
455: 
456: 
457:     return np.asarray(
458:         output,
459:         dtype=np.float64

143: class RecurrentForecaster(nn.Module):
144: 
145:     def __init__(
146:         self,
147:         model_type,
148:         future_length
149:     ):
150: 
151:         super().__init__()
152: 
153:         self.model_type = model_type
154: 
155:         self.future_length = future_length
156: 
157: 
158:         if model_type == "LSTM":
159: 
160:             self.encoder = nn.LSTM(
161: 
162:                 input_size=2,
163: 
164:                 hidden_size=HIDDEN_SIZE,
165: 
166:                 num_layers=NUM_LAYERS,
167: 
168:                 batch_first=True,
169: 
170:                 dropout=DROPOUT
171:             )
172: 
173: 
174:         elif model_type == "GRU":
175: 
176:             self.encoder = nn.GRU(
177: 
178:                 input_size=2,
179: 
180:                 hidden_size=HIDDEN_SIZE,
181: 
182:                 num_layers=NUM_LAYERS,
183: 
184:                 batch_first=True,
185: 
186:                 dropout=DROPOUT
187:             )
188: 
189: 
190:         else:
191: 
192:             raise ValueError(
193:                 model_type
194:             )
195: 
196: 
197:         self.head = nn.Sequential(
198: 
199:             nn.Linear(
200:                 HIDDEN_SIZE,
201:                 HIDDEN_SIZE
202:             ),
203: 
204:             nn.ReLU(),
205: 
206:             nn.Linear(
207:                 HIDDEN_SIZE,
208:                 future_length * 2
209:             )
210:         )
211: 
212: 
213:     def forward(self, x):
214: 
215: 
216:         if self.model_type == "LSTM":
217: 
218:             _, (hidden, _) = self.encoder(
219:                 x
220:             )
221: 
222:         else:
223: 
224:             _, hidden = self.encoder(
225:                 x
226:             )
227: 
228: 
229:         latent = hidden[
230:             -1
231:         ]
232: 
233: 
234:         output = self.head(
235:             latent
236:         )
237: 
238: 
239:         return output.view(
240: 
241:             -1,
242: 
243:             self.future_length,
244: 
245:             2
246:         )

249: # ============================================================
250: # WEIGHTED LOSS
251: # ============================================================
252: 
253: def weighted_smooth_l1(
254:     prediction,
255:     target,
256:     weights
257: ):
258: 
259:     # loss per coordinate
260:     element_loss = nn.functional.smooth_l1_loss(
261: 
262:         prediction,
263: 
264:         target,
265: 
266:         reduction="none"
267: 
268:     )
269: 
270: 
271:     # one loss per sample
272:     sample_loss = element_loss.mean(
273:         dim=(1, 2)
274:     )
275: 
276: 
277:     weighted = (
278: 
279:         sample_loss
280:         *
281:         weights
282: 
283:     )
284: 
285: 
286:     return (
287: 
288:         weighted.sum()
289: 
290:         /
291:         weights.sum()
292: 
293:     )

296: # ============================================================
297: # INVERSE CANONICAL TRANSFORM
298: # ============================================================
299: 
300: def inverse_transform_batch(
301:     canonical,
302:     origin,
303:     angle,
304:     scale
305: ):
306: 
307:     output = []
308: 
309: 
310:     for i in range(
311:         len(canonical)
312:     ):
313: 
314: 
315:         a = float(
316:             angle[i]
317:         )
318: 
319: 
320:         c = np.cos(a)
321: 
322:         s = np.sin(a)
323: 
324: 
325:         R = np.array(
326: 
327:             [
328:                 [c, s],
329:                 [-s, c]
330:             ],
331: 
332:             dtype=np.float64
333:         )
334: 
335: 
336:         rotated = (
337: 
338:             canonical[i]
339:             *
340:             float(
341:                 scale[i]
342:             )
343: 
344:         )
345: 
346: 
347:         # canonical = local @ R.T / scale
348:         # therefore local = canonical*scale @ R
349: 
350:         local = (
351:             rotated
352:             @
353:             R
354:         )
355: 
356: 
357:         absolute = (
358: 
359:             local
360:             +
361:             origin[i]
362: 
363:         )
364: 
365: 
366:         output.append(
367:             absolute
368:         )
369: 
370: 
371:     return np.asarray(
372:         output,
373:         dtype=np.float64
374:     )

377: # ============================================================
378: # BASIC TRAJECTORY ERRORS
379: # ============================================================
380: 
381: def sample_ade_fde(
382:     prediction,
383:     truth
384: ):
385: 
386:     errors = np.linalg.norm(
387: 
388:         prediction
389:         -
390:         truth,
391: 
392:         axis=2
393: 
394:     )
395: 
396: 
397:     ADE = errors.mean(
398:         axis=1
399:     )
400: 
401:     FDE = errors[
402:         :,
403:         -1
404:     ]
405: 
406: 
407:     return ADE, FDE
